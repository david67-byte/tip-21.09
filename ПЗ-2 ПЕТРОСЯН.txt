// задание 1
#include <iostream>
using namespace std;

int main() {
    int a = 4;
    int b = 5;

    cout << "До: a = " << a << ", b = " << b << endl;
    a = a + b;
    b = a - b;
    a = a - b;

    cout << "После: a = " << a << ", b = " << b << endl;

    return 0;
}


// задание 2
#include <iostream>
using namespace std;

int main() {
    int n;
    cout << "Введите n: ";
    cin >> n;

    if (n < 0) {
        cout << "Факториал для отрицательных чисел не определён." << endl;
        return 0;
    }


    long long current_factorial = 1; 
    long long result = 1;             

    for (int i = 1; i <= n; ++i) {

        current_factorial *= i;

        result *= current_factorial;
    }

    cout << "F(" << n << ") = " << result << endl;

    return 0;
}

#include <iostream>           

int main() {                  
    int A = 0;                
    int B;                    

    B = 2 - (A & 1);          // Вычисляем B: если A = 0, то A & 1 = 0 → B = 2;
                              //   если A = 1, то A & 1 = 1 → B = 1.
                              //   (A & 1) — побитовое И, выделяет младший бит A.

    std::cout << "A = " << A  // Выводим значение A на экран
              << ", B = " << B// Выводим вычисленное значение B
              << std::endl;   // Переходим на новую строку

    return 0;
}
